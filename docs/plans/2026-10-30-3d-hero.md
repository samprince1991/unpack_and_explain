# 3D Hero Scene Implementation Plan

> **For agentic workers:** Execute inline with superpowers:executing-plans (CLAUDE.md: no subagents). Steps use checkbox (`- [ ]`) syntax for tracking. **Do not commit** unless the user asks (CLAUDE.md); each task ends with a checkpoint instead.

**Goal:** Every new reel's mechanism is a three.js hero scene, built from a reusable, brand-safe 3D style library and enforced by the script validator.

**Architecture:** A pure camera module (`stage.ts`) turns the frame into a view (rotation, pan, zoom) and projects 3D points to frame pixels. `Stage3D` renders that view as nested `<group>`s inside a fixed-camera `ThreeCanvas`, so the 3D and the 2D labels (`Label3D`) always agree. Four styles (`WireObject`, `ExplodedView`, `NetworkGraph3D`, `Bars3D`) draw with drei `Edges` / `Line` in token colors; their motion maths lives in pure, unit-tested helpers. `PostScript.hero3d` plus a repo guard test make the 3D mechanism mandatory for new posts.

**Tech Stack:** Remotion 4.0.539, `@remotion/three`, `@react-three/fiber` 9, `three` 0.187, `@react-three/drei` 10, vitest.

Spec: `docs/specs/2026-10-30-3d-hero-design.md`.

**Deviations from the spec (decided while planning):**
- Camera is perspective only (no `ortho`); YAGNI until a post needs isometric.
- Shots use `pan: [x, y]` (view-space offset) instead of a 3D `target`, because the scene orbits under the camera. Same idea as 2D `Camera` shots.
- `useStageView()` returns a `StageView` that is passed to both `Stage3D` and `Label3D`, so labels track the 3D exactly.
- Post 18 is a carousel (no `script.ts`), so only posts 01, 02 and 20 opt out.

All commands run from `video/`.

```ts
import { interpolate } from "remotion";
import { Euler, PerspectiveCamera, Vector3 } from "three";
import { layout } from "../../brand/layout";
import { clamp, ease, idle } from "../../brand/motion";

export type Vec3 = [number, number, number];

export type Shot3D = {
  // Local frame the camera arrives (usually a useCue value).
  at: number;
  // View-space point to center (scene units, after the orbit), and zoom (1 = none).
  pan: [number, number];
  zoom: number;
};

export type StageOptions = {
  // Degrees per second around the vertical axis, and the starting angle.
  orbit?: number;
  angle?: number;
  // Degrees the scene leans toward the camera.
  tilt?: number;
  shots?: Shot3D[];
  // Frames to travel into each shot.
  travel?: number;
  // Slow push-in across the section (1 = off).
  pushIn?: number;
  // Frames the push-in spans: useSection().durationInFrames in reels.
  duration: number;
  fps: number;
};

export type StageView = { rotation: Vec3; pan: [number, number]; zoom: number };

// The 3D canvas fills the visual region on the centered column (same box as WireGlobe).
export const STAGE = {
  left: layout.column.x,
  top: layout.regions.visual.top,
  width: layout.column.width,
  height: layout.regions.visual.bottom - layout.regions.visual.top,
  fov: 40,
  distance: 6,
} as const;

const rad = (deg: number) => (deg * Math.PI) / 180;
const lerp = (a: number, b: number, t: number) => a + (b - a) * t;

// Same glide logic as the 2D Camera: the latest shot whose move has started wins.
export const cameraAt = (frame: number, options: StageOptions): StageView => {
  const {
    orbit = 0,
    angle = 0,
    tilt = 15,
    shots = [],
    travel = 24,
    pushIn = idle.pushIn,
    duration,
    fps,
  } = options;
  const all: Shot3D[] = [{ at: 0, pan: [0, 0], zoom: 1 }, ...shots];
  let from = all[0];
  let to = all[0];
  let t = 1;
  for (let i = 1; i < all.length; i++) {
    if (frame >= all[i].at - travel) {
      from = all[i - 1];
      to = all[i];
      t = interpolate(frame, [all[i].at - travel, all[i].at], [0, 1], {
        ...clamp,
        easing: ease.smooth,
      });
    }
  }
  const push = interpolate(frame, [0, Math.max(1, duration)], [1, pushIn], clamp);
  return {
    rotation: [rad(tilt), rad(angle + (orbit * frame) / fps), 0],
    pan: [lerp(from.pan[0], to.pan[0], t), lerp(from.pan[1], to.pan[1], t)],
    zoom: lerp(from.zoom, to.zoom, t) * push,
  };
};

const camera = new PerspectiveCamera(STAGE.fov, STAGE.width / STAGE.height, 0.1, 100);
camera.position.set(0, 0, STAGE.distance);
camera.updateMatrixWorld();

// Frame pixels of a scene point under `view`. Mirrors Stage3D's groups:
// rotate, then pan and zoom, then the fixed camera.
export const projectToScreen = (point: Vec3, view: StageView) => {
  const v = new Vector3(...point).applyEuler(new Euler(...view.rotation));
  v.set((v.x - view.pan[0]) * view.zoom, (v.y - view.pan[1]) * view.zoom, v.z * view.zoom);
  v.project(camera);
  return {
    x: STAGE.left + ((v.x + 1) / 2) * STAGE.width,
    y: STAGE.top + ((1 - v.y) / 2) * STAGE.height,
  };
};
```

```tsx
import { ThreeCanvas } from "@remotion/three";
import { useCurrentFrame, useVideoConfig } from "remotion";
import { STAGE, cameraAt, type StageOptions, type StageView } from "./stage";

// The frame's view: pass the same object to Stage3D and every Label3D.
export const useStageView = (options: Omit<StageOptions, "fps">): StageView => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  return cameraAt(frame, { ...options, fps });
};

// Canvas for 3D hero scenes: fills the visual region on the 540 column. The camera
// never moves; the scene groups do (see stage.ts projectToScreen). Animate only from
// useCurrentFrame(), never R3F useFrame().
export const Stage3D: React.FC<{ view: StageView; children: React.ReactNode }> = ({
  view,
  children,
}) => (
  <ThreeCanvas
    width={STAGE.width}
    height={STAGE.height}
    camera={{ position: [0, 0, STAGE.distance], fov: STAGE.fov }}
    style={{ position: "absolute", left: STAGE.left, top: STAGE.top }}
  >
    <group
      position={[-view.pan[0] * view.zoom, -view.pan[1] * view.zoom, 0]}
      scale={view.zoom}
    >
      <group rotation={view.rotation}>{children}</group>
    </group>
  </ThreeCanvas>
);
```

```tsx
import { Edges } from "@react-three/drei";
import { interpolate, useCurrentFrame } from "remotion";
import { clamp, durations, ease } from "../../brand/motion";
import { FILL, line, type LineRole } from "./materials";
import type { Vec3 } from "./stage";

export type WireShape = "box" | "cylinder" | "slab" | "card" | "phone";

// Default sizes (scene units): card is ISO card ratio, phone a slim slab.
const SIZES: Record<WireShape, Vec3> = {
  box: [1, 1, 1],
  cylinder: [1, 1, 1],
  slab: [1.6, 0.12, 1.2],
  card: [1.6, 1, 0.04],
  phone: [1, 2, 0.12],
};

type WireObjectProps = {
  shape: WireShape;
  size?: Vec3;
  position?: Vec3;
  rotation?: Vec3;
  // Non-uniform scale, e.g. a bar rising (keep > 0).
  scale?: Vec3;
  role?: LineRole;
  // Frame the object scales and fades in.
  drawAt?: number;
  opacity?: number;
};

// A primitive as line art: dark solid fill (hides back edges) + token-colored edges.
export const WireObject: React.FC<WireObjectProps> = ({
  shape,
  size = SIZES[shape],
  position = [0, 0, 0],
  rotation = [0, 0, 0],
  scale = [1, 1, 1],
  role = "stroke",
  drawAt = 0,
  opacity = 1,
}) => {
  const frame = useCurrentFrame();
  const p = interpolate(frame, [drawAt, drawAt + durations.enter], [0, 1], {
    ...clamp,
    easing: ease.out,
  });
  const grow = 0.85 + 0.15 * p;
  const alpha = p * opacity;
  return (
    <mesh
      position={position}
      rotation={rotation}
      scale={[scale[0] * grow, scale[1] * grow, scale[2] * grow]}
      visible={alpha > 0}
    >
      {shape === "cylinder" ? (
        // 16 sides: facets show as a few vertical lines, like a hand-drawn can.
        <cylinderGeometry args={[size[0] / 2, size[0] / 2, size[1], 16]} />
      ) : (
        <boxGeometry args={size} />
      )}
      <meshBasicMaterial
        color={FILL}
        transparent
        opacity={alpha}
        polygonOffset
        polygonOffsetFactor={1}
        polygonOffsetUnits={1}
      />
      <Edges threshold={15} {...line(role)} transparent opacity={alpha} />
    </mesh>
  );
};
```

```ts
import { interpolate } from "remotion";
import { clamp, ease } from "../../brand/motion";

export type ExplodeOptions = {
  count: number;
  // Layer spacing when stacked, and the extra gap once exploded (scene units).
  thickness: number;
  gap: number;
  explodeAt: number;
  explodeFrames: number;
  // Frames between layers starting to move.
  stagger: number;
};

// Offset of layer i along the axis, centered on the middle of the stack.
export const layerOffset = (i: number, frame: number, o: ExplodeOptions) => {
  const start = o.explodeAt + i * o.stagger;
  const p = interpolate(frame, [start, start + o.explodeFrames], [0, 1], {
    ...clamp,
    easing: ease.smooth,
  });
  return (i - (o.count - 1) / 2) * (o.thickness + o.gap * p);
};
```

```tsx
import { useCurrentFrame } from "remotion";
import { durations } from "../../brand/motion";
import { layerOffset } from "./explode";
import type { Vec3 } from "./stage";
import { WireObject, type WireShape } from "./WireObject";

export type Layer = { shape: WireShape; size?: Vec3 };

type ExplodedViewProps = {
  layers: Layer[];
  // The one amber layer (one index, so two can't be amber).
  keyIndex?: number;
  axis?: "y" | "z";
  thickness?: number;
  gap?: number;
  // Layers draw in one after another from drawAt, then separate at explodeAt.
  drawAt?: number;
  explodeAt: number;
  explodeFrames?: number;
  stagger?: number;
};

// Scene position of layer i: use with Label3D to label a layer.
export const layerPosition = (
  i: number,
  frame: number,
  { layers, axis = "y", thickness = 0.12, gap = 0.5, explodeAt, explodeFrames = 20, stagger = 0 }: ExplodedViewProps,
): Vec3 => {
  const d = layerOffset(i, frame, { count: layers.length, thickness, gap, explodeAt, explodeFrames, stagger });
  return axis === "y" ? [0, d, 0] : [0, 0, d];
};

// "What's inside": a stack that pulls apart so the key layer can be seen.
export const ExplodedView: React.FC<ExplodedViewProps> = (props) => {
  const frame = useCurrentFrame();
  const { layers, keyIndex, drawAt = 0 } = props;
  return (
    <>
      {layers.map((layer, i) => (
        <WireObject
          key={i}
          shape={layer.shape}
          size={layer.size}
          position={layerPosition(i, frame, props)}
          role={i === keyIndex ? "amber" : "stroke"}
          drawAt={drawAt + i * durations.stagger}
        />
      ))}
    </>
  );
};
```

```ts
import { interpolate } from "remotion";
import { clamp, ease } from "../../brand/motion";
import type { Vec3 } from "./stage";

export type Node3D = { id: string; pos: Vec3; shape?: "dot" | "box" };

const dist = (a: Vec3, b: Vec3) => Math.hypot(b[0] - a[0], b[1] - a[1], b[2] - a[2]);

export const nodePos = (nodes: Node3D[], id: string): Vec3 => {
  const node = nodes.find((n) => n.id === id);
  if (!node) throw new Error(`Unknown network node "${id}"`);
  return node.pos;
};

// Where the packet is along the route: constant speed over total length.
export const packetAt = (
  frame: number,
  route: string[],
  nodes: Node3D[],
  travelAt: number,
  travelFrames: number,
) => {
  const pts = route.map((id) => nodePos(nodes, id));
  const progress = interpolate(frame, [travelAt, travelAt + travelFrames], [0, 1], {
    ...clamp,
    easing: ease.smooth,
  });
  const lengths = pts.slice(1).map((p, i) => dist(pts[i], p));
  let left = progress * lengths.reduce((s, l) => s + l, 0);
  for (let i = 0; i < lengths.length; i++) {
    if (left <= lengths[i] || i === lengths.length - 1) {
      const t = lengths[i] === 0 ? 1 : Math.min(1, left / lengths[i]);
      const [a, b] = [pts[i], pts[i + 1]];
      const pos: Vec3 = [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t];
      return { pos, segment: i, progress };
    }
    left -= lengths[i];
  }
  return { pos: pts[0], segment: 0, progress };
};
```

Note: tests 1-2 sit outside the travel window, where `clamp` makes progress exactly 0 / 1, so exact matches are safe. Test 1 expects `segment: 0, pos: [0,0,0]` at progress 0: `left = 0 <= lengths[0]` → t = 0 → pos a. Test 2: progress 1, left = 2; i=0: 2 > 1 → left = 1; i=1: 1 <= 1 → t = 1 → c. Run → PASS.

```tsx
  Array.from({ length: n + 1 }, (_, i) => {

    return [Math.cos(a) * r, Math.sin(a) * r, 0];

  route.some((id, i) => i > 0 && ((route[i - 1] === a && id === b) || (route[i - 1] === b && id === a)));

  travelFrames = 60,

  const packet = travelling ? packetAt(frame, route, nodes, travelAt, travelFrames) : null;
  const tail = travelling ? packetAt(frame - 5, route, nodes, travelAt, travelFrames) : null;

        const from = nodePos(nodes, a);
        const to = nodePos(nodes, b);

        const p = interpolate(frame, [start, start + durations.draw], [0, 1], { ...clamp, easing: ease.out });
        if (p === 0) return null;
        const end: Vec3 = [from[0] + (to[0] - from[0]) * p, from[1] + (to[1] - from[1]) * p, from[2] + (to[2] - from[2]) * p];
        const lit = travelling && onRoute(route, [a, b]) && frame >= travelAt;
        return <Line key={`${a}-${b}`} points={[from, end]} {...line(lit ? "soft" : "faint")} />;
      })}
      {nodes.map((n, i) =>

            <meshBasicMaterial color={lineColor("amber")} />
          </mesh>
        </>
```

```ts
import { interpolate } from "remotion";
import { clamp, ease } from "../../brand/motion";

export type BarOptions = {
  values: number[];
  maxHeight: number;
  riseAt: number;
  stagger: number;
  riseFrames: number;
};

export const barHeight = (i: number, frame: number, o: BarOptions) => {
  const start = o.riseAt + i * o.stagger;
  const p = interpolate(frame, [start, start + o.riseFrames], [0, 1], { ...clamp, easing: ease.out });
  return (o.values[i] / Math.max(...o.values)) * o.maxHeight * p;
};

export const barX = (i: number, count: number, spacing: number) => (i - (count - 1) / 2) * spacing;
```

```tsx
import { Line } from "@react-three/drei";
import { useCurrentFrame } from "remotion";
import { durations } from "../../brand/motion";
import { barHeight, barX } from "./bars";
import { line } from "./materials";
import type { Vec3 } from "./stage";
import { WireObject } from "./WireObject";

type Bars3DProps = {
  values: number[];
  // The one amber column (the result).
  keyIndex?: number;
  riseAt: number;
  stagger?: number;
  maxHeight?: number;
  spacing?: number;
  width?: number;
};

// Top-center of bar i at this frame: use with Label3D for the value.
export const barTop = (i: number, frame: number, p: Bars3DProps): Vec3 => {
  const { values, riseAt, stagger = durations.stagger, maxHeight = 2, spacing = 0.6 } = p;
  const h = barHeight(i, frame, { values, maxHeight, riseAt, stagger, riseFrames: durations.draw });
  return [barX(i, values.length, spacing), h - maxHeight / 2, 0];
};

// Columns rising from a faint floor grid; the floor sits at y = -maxHeight / 2 so the
// chart is centered on the stage.
export const Bars3D: React.FC<Bars3DProps> = (props) => {
  const frame = useCurrentFrame();
  const { values, keyIndex, maxHeight = 2, spacing = 0.6, width = 0.4 } = props;
  const floor = -maxHeight / 2;
  const half = (values.length * spacing) / 2;
  const grid: Vec3[] = [];
  for (let k = -2; k <= 2; k++) {
    const z = (k * half) / 2;
    const x = (k * half) / 2;
    grid.push([-half, floor, z], [half, floor, z], [x, floor, -half], [x, floor, half]);
  }
  return (
    <>
      <Line points={grid} segments {...line("faint")} />
      {values.map((_, i) => {
        const top = barTop(i, frame, props);
        const h = Math.max(0.001, top[1] - floor);
        return (
          <WireObject
            key={i}
            shape="box"
            size={[width, 1, width]}
            position={[top[0], floor + h / 2, 0]}
            scale={[1, h, 1]}
            role={i === keyIndex ? "amber" : "stroke"}
          />
        );
      })}
    </>
  );
};
```
