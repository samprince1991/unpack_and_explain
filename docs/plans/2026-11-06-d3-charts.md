# D3 Charts Implementation Plan

> **For agentic workers:** Execute inline with superpowers:executing-plans (this repo forbids subagents, see CLAUDE.md). Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A branded, frame-driven chart kit (bar, line, pie/donut, treemap, tree, sankey, funnel) for reel scenes and carousel slides, with D3 doing the layout maths.

**Architecture:** Each chart has a pure `layout.ts` (D3 scales/shapes/hierarchy/sankey → plain shapes, unit-tested, no React, no frame) and a `<Name>.tsx` that draws those shapes as SVG and animates them from `useCurrentFrame()`. Components never import D3 directly. Shared rules live in `src/brand/charts.ts`; shared helpers in `src/components/charts/geometry.ts` (pure) and `common.tsx` (React). `still` gives the final pose for carousels.

**Tech Stack:** Remotion 4.0.539, React 19, TypeScript, vitest, `d3-scale` 4, `d3-shape` 3, `d3-array` 3, `d3-hierarchy` 3, `d3-sankey` 0.12, `@remotion/paths`.

**Spec:** `docs/specs/2026-11-06-d3-charts-design.md`. Refinements made while planning (Task 11 writes them back into the spec):
- Value labels are controlled by `values: "none" | "key" | "all"` (default `"key"`). Names (`name` on a datum) are always drawn when given and are fit-checked; a name that can't fit throws.
- Limits tightened for readability at 46 px labels: tree <= 5 nodes per level, sankey <= 4 nodes per column.
- Count-up is built in (bar values count with the bar's growth); `NumberCounter` is not used.

**Commits:** CLAUDE.md says don't commit unless the user asks. Each task ends with a commit step: run it only if the user has OK'd commits for this work; otherwise skip and leave the changes staged-ready.

```ts
import { charts } from "../../brand/charts";

// Pure chart geometry helpers: no React, no frame. Layouts import only this file and D3.
export type Box = { x: number; y: number; width: number; height: number };
export type Point = { x: number; y: number };

// One label line including spacing.
export const LINE = charts.labelSize * 1.4;

// Thrown when data can't be drawn readably. Fails loudly in Studio instead of shipping.
export class ChartLimitError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ChartLimitError";
  }
}

export const checkCount = (chart: string, what: string, count: number, max: number): void => {
  if (count === 0) {
    throw new ChartLimitError(`${chart}: no ${what}`);
  }
  if (count > max) {
    throw new ChartLimitError(
      `${chart}: ${count} ${what}, max ${max}. Group the smallest into "Other".`,
    );
  }
};

export const checkValues = (chart: string, values: number[], allowNegative = false): void => {
  const bad = values.find((v) => !Number.isFinite(v) || (!allowNegative && v < 0));
  if (bad !== undefined) {
    throw new ChartLimitError(`${chart}: values must be finite${allowNegative ? "" : " and >= 0"}, got ${bad}`);
  }
};

export const uniqueIds = (chart: string, ids: string[]): void => {
  const seen = new Set<string>();
  for (const id of ids) {
    if (seen.has(id)) {
      throw new ChartLimitError(`${chart}: duplicate id "${id}"`);
    }
    seen.add(id);
  }
};

// Rough rendered width of a label in px.
export const textWidth = (text: string, size: number = charts.labelSize): number =>
  text.length * size * charts.charWidth;

export const checkFits = (chart: string, text: string, room: number): void => {
  const width = textWidth(text);
  if (width > room) {
    throw new ChartLimitError(
      `${chart}: "${text}" is ~${Math.round(width)} px wide, room for ${Math.round(room)} px. Shorten it.`,
    );
  }
};

export const formatValue = (value: number, format?: (v: number) => string, unit?: string): string => {
  const text = format ? format(value) : Math.round(value).toLocaleString("en-US");
  if (!unit) {
    return text;
  }
  return unit === "%" ? `${text}%` : `${text} ${unit}`;
};
```

```ts
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative } from "node:path";
import { describe, expect, it } from "vitest";

// D3 is for maths only. These modules mutate the DOM or run on the wall clock,
// which flickers in Remotion renders (see docs/specs/2026-11-06-d3-charts-design.md).
const BANNED = /from\s+["'](d3|d3-selection|d3-transition|d3-timer|d3-ease)["']/;

const SRC = join(process.cwd(), "src");
const walk = (dir: string): string[] =>
  readdirSync(dir).flatMap((name) => {
    const full = join(dir, name);
    return statSync(full).isDirectory() ? walk(full) : [full];
  });

describe("d3 import guard", () => {
  it("only imports frame-safe d3 modules", () => {
    const bad = walk(SRC)
      .filter((f) => /\.(ts|tsx)$/.test(f))
      .filter((f) => BANNED.test(readFileSync(f, "utf8")))
      .map((f) => relative(SRC, f));
    expect(bad).toEqual([]);
  });
});
```

```tsx
import { interpolateColors } from "remotion";
import { charts } from "../../brand/charts";
import { fonts, weights } from "../../brand/fonts";
import { durations } from "../../brand/motion";
import { colors } from "../../brand/tokens";
import { enterProgress, exitProgress, stagger } from "../motion/progress";
import type { Box } from "./geometry";

export type ValueLabels = "none" | "key" | "all";

// Local frames. `still` = carousel: final pose, key already amber, no animation.
export type ChartTiming = { at?: number; keyAt?: number; exitAt?: number; still?: boolean };

export type ChartBaseProps = ChartTiming & {
  box: Box;
  // The one amber item. Omit for no amber.
  keyId?: string;
  // Which value labels show. "key" = only the key item's, appearing as it turns amber.
  values?: ValueLabels;
  format?: (v: number) => string;
  unit?: string;
};

export type Progress = { enter: number; key: number; exit: number };

// enter: item `index`'s staggered entrance; key: amber on at keyAt (on from the start when
// keyAt is unset); exit: fade from exitAt.
export const chartProgress = (
  frame: number,
  timing: ChartTiming,
  index = 0,
  frames: number = durations.enter,
): Progress => {
  if (timing.still) {
    return { enter: 1, key: 1, exit: 1 };
  }
  return {
    enter: enterProgress(frame, stagger(timing.at ?? 0, index, charts.stagger), frames),
    key: timing.keyAt === undefined ? 1 : enterProgress(frame, timing.keyAt),
    exit: exitProgress(frame, timing.exitAt),
  };
};

// The key item's color fades from `from` to amber as `key` goes 0 -> 1.
export const keyColor = (isKey: boolean, key: number, from: string): string =>
  isKey ? interpolateColors(key, [0, 1], [from, colors.amber[500]]) : from;

export const valueOpacity = (values: ValueLabels, isKey: boolean, p: Progress): number => {
  if (values === "all") {
    return p.enter;
  }
  if (values === "key" && isKey) {
    return p.enter * p.key;
  }
  return 0;
};

// Text sitting on a fill: switches to the dark base once the fill has turned amber.
export const textOn = (isKey: boolean, key: number): string =>
  isKey && key > 0.5 ? colors.bg.base : colors.text.secondary;

type ChartTextProps = {
  x: number;
  y: number;
  text: string;
  anchor?: "start" | "middle" | "end";
  color?: string;
  opacity?: number;
  size?: number;
  // Poppins (big numbers) instead of Inter.
  heading?: boolean;
};

export const ChartText: React.FC<ChartTextProps> = ({
  x,
  y,
  text,
  anchor = "middle",
  color = colors.text.secondary,
  opacity = 1,
  size = charts.labelSize,
  heading = false,
}) =>
  opacity <= 0 ? null : (
    <text
      x={x}
      y={y}
      textAnchor={anchor}
      dominantBaseline="central"
      fontFamily={heading ? fonts.heading : fonts.body}
      fontWeight={heading ? weights.heading : weights.bodyMedium}
      fontSize={size}
      fill={color}
      opacity={opacity}
      style={{ fontVariantNumeric: "tabular-nums" }}
    >
      {text}
    </text>
  );
```

```ts
import { max } from "d3-array";
import { scaleBand, scaleLinear } from "d3-scale";
import { charts } from "../../../brand/charts";
import {
  type Box,
  ChartLimitError,
  checkCount,
  checkFits,
  checkValues,
  LINE,
  type Point,
  uniqueIds,
} from "../geometry";

export type BarDatum = { id: string; name?: string; value: number };
export type Orientation = "vertical" | "horizontal";
export type BarShape = BarDatum & {
  x: number;
  y: number;
  width: number;
  height: number;
  valueAt: Point;
  nameAt: Point;
};
export type BarLayout = {
  bars: BarShape[];
  baseline: { x1: number; y1: number; x2: number; y2: number };
};

const CHART = "BarChart";

// Vertical: names under the bars, values above. Horizontal: name above each bar, value
// after its end. `top` fixes the value axis maximum (default: the largest value).
export const barLayout = (
  data: BarDatum[],
  box: Box,
  orientation: Orientation = "vertical",
  top?: number,
): BarLayout => {
  checkCount(CHART, "bars", data.length, charts.limits.bars);
  checkValues(CHART, data.map((d) => d.value));
  uniqueIds(CHART, data.map((d) => d.id));
  const ids = data.map((d) => d.id);
  const domainMax = top ?? max(data, (d) => d.value) ?? 0;
  const hasNames = data.some((d) => d.name);

  if (orientation === "vertical") {
    const baseY = box.y + box.height - (hasNames ? LINE : 0);
    const band = scaleBand(ids, [box.x, box.x + box.width]).padding(charts.barGap);
    const y = scaleLinear([0, domainMax || 1], [baseY, box.y + LINE]);
    const bars = data.map((d) => {
      const x = band(d.id) ?? box.x;
      const cx = x + band.bandwidth() / 2;
      if (d.name) {
        checkFits(CHART, d.name, band.step());
      }
      return {
        ...d,
        x,
        y: y(d.value),
        width: band.bandwidth(),
        height: baseY - y(d.value),
        valueAt: { x: cx, y: y(d.value) - LINE / 2 },
        nameAt: { x: cx, y: baseY + LINE / 2 },
      };
    });
    return { bars, baseline: { x1: box.x, y1: baseY, x2: box.x + box.width, y2: baseY } };
  }

  const valueRoom = charts.labelSize * 3.5;
  const band = scaleBand(ids, [box.y, box.y + box.height]).paddingInner(0.15);
  const x = scaleLinear([0, domainMax || 1], [box.x, box.x + box.width - valueRoom]);
  // Tighter than LINE so horizontal bars still fit a carousel visual region.
  const nameLine = hasNames ? charts.labelSize * 1.15 : 0;
  const thick = Math.min(band.bandwidth() - nameLine, charts.barMax);
  if (thick < charts.labelSize * 0.5) {
    throw new ChartLimitError(`${CHART}: bars too thin; use fewer bars or a taller box`);
  }
  const bars = data.map((d) => {
    const slot = band(d.id) ?? box.y;
    const by = slot + nameLine;
    if (d.name) {
      checkFits(CHART, d.name, box.width);
    }
    return {
      ...d,
      x: box.x,
      y: by,
      width: x(d.value) - box.x,
      height: thick,
      valueAt: { x: x(d.value) + 16, y: by + thick / 2 },
      nameAt: { x: box.x, y: slot + nameLine / 2 },
    };
  });
  return { bars, baseline: { x1: box.x, y1: box.y, x2: box.x, y2: box.y + box.height } };
};
```

```tsx
import { Interactive, useCurrentFrame } from "remotion";
import { charts } from "../../../brand/charts";
import { colors } from "../../../brand/tokens";
import { type ChartBaseProps, ChartText, chartProgress, keyColor, valueOpacity } from "../common";
import { formatValue } from "../geometry";
import { type BarDatum, barLayout, type Orientation } from "./layout";

export type BarChartProps = ChartBaseProps & {
  data: BarDatum[];
  orientation?: Orientation;
  // Fixed value-axis maximum (default: the largest value).
  max?: number;
};

// Bars grow from the baseline one after another; values count up with them. The key bar
// turns amber at keyAt. SVG, for SceneCanvas / SlideCanvas.
export const BarChart: React.FC<BarChartProps> = ({
  data,
  box,
  orientation = "vertical",
  max,
  keyId,
  values = "key",
  format,
  unit,
  ...timing
}) => {
  const frame = useCurrentFrame();
  const { bars, baseline } = barLayout(data, box, orientation, max);
  const base = chartProgress(frame, timing);
  const vertical = orientation === "vertical";

  return (
    <Interactive.G name="Bar chart" style={{ opacity: base.exit }}>
      <line
        {...baseline}
        stroke={colors.diagram.faint}
        strokeWidth={charts.strokeSoft}
        strokeLinecap="round"
        opacity={base.enter}
      />
      {bars.map((b, i) => {
        const p = chartProgress(frame, timing, i + 1);
        const isKey = b.id === keyId;
        const w = vertical ? b.width : b.width * p.enter;
        const h = vertical ? b.height * p.enter : b.height;
        const y = vertical ? b.y + b.height - h : b.y;
        const valueX = vertical ? b.valueAt.x : b.x + w + 16;
        const valueY = vertical ? y - (b.y - b.valueAt.y) : b.valueAt.y;
        return (
          <g key={b.id}>
            {w > 0 && h > 0 ? (
              <rect
                x={b.x}
                y={y}
                width={w}
                height={h}
                rx={Math.min(charts.radius, w / 2, h / 2)}
                fill={keyColor(isKey, p.key, colors.diagram.fill)}
                stroke={keyColor(isKey, p.key, colors.diagram.soft)}
                strokeWidth={charts.stroke}
              />
            ) : null}
            {b.name ? (
              <ChartText
                x={b.nameAt.x}
                y={b.nameAt.y}
                text={b.name}
                anchor={vertical ? "middle" : "start"}
                color={colors.text.muted}
                opacity={p.enter}
              />
            ) : null}
            <ChartText
              x={valueX}
              y={valueY}
              text={formatValue(b.value * p.enter, format, unit)}
              anchor={vertical ? "middle" : "start"}
              color={keyColor(isKey, p.key, colors.text.secondary)}
              opacity={valueOpacity(values, isKey, p)}
            />
          </g>
        );
      })}
    </Interactive.G>
  );
};
```

```ts
import { max, min } from "d3-array";
import { scaleLinear, scalePoint } from "d3-scale";
import { curveMonotoneX, line } from "d3-shape";
import { charts } from "../../../brand/charts";
import {
  type Box,
  ChartLimitError,
  checkCount,
  checkFits,
  checkValues,
  LINE,
  type Point,
  uniqueIds,
} from "../geometry";

export type LineDatum = { id: string; name?: string; value: number };
export type LinePointShape = LineDatum & { x: number; y: number; valueAt: Point; nameAt: Point };
export type LineLayout = {
  d: string;
  points: LinePointShape[];
  baseline: { x1: number; y1: number; x2: number; y2: number };
};

const CHART = "LineChart";

// Names along the bottom, values above the points. `range` fixes the value axis
// (default: 0 or the lowest value, up to the highest).
export const lineLayout = (data: LineDatum[], box: Box, range?: [number, number]): LineLayout => {
  checkCount(CHART, "points", data.length, charts.limits.points);
  if (data.length < 2) {
    throw new ChartLimitError(`${CHART}: needs at least 2 points`);
  }
  const values = data.map((d) => d.value);
  checkValues(CHART, values, true);
  uniqueIds(CHART, data.map((d) => d.id));

  const hasNames = data.some((d) => d.name);
  const baseY = box.y + box.height - (hasNames ? LINE : 0);
  const x = scalePoint(data.map((d) => d.id), [box.x, box.x + box.width]).padding(0.5);
  const lo = range?.[0] ?? Math.min(0, min(values) ?? 0);
  const hi = range?.[1] ?? (max(values) ?? 1);
  const y = scaleLinear([lo, hi === lo ? lo + 1 : hi], [baseY - charts.labelSize * 0.5, box.y + LINE]);

  const points = data.map((d) => {
    const px = x(d.id) ?? box.x;
    const py = y(d.value);
    if (d.name) {
      checkFits(CHART, d.name, x.step());
    }
    return { ...d, x: px, y: py, valueAt: { x: px, y: py - LINE * 0.6 }, nameAt: { x: px, y: baseY + LINE / 2 } };
  });
  const d =
    line<LinePointShape>()
      .x((p) => p.x)
      .y((p) => p.y)
      .curve(curveMonotoneX)(points) ?? "";
  return { d, points, baseline: { x1: box.x, y1: baseY, x2: box.x + box.width, y2: baseY } };
};
```

```ts
import { sum } from "d3-array";
import { arc, pie } from "d3-shape";
import { charts } from "../../../brand/charts";
import {
  type Box,
  ChartLimitError,
  checkCount,
  checkValues,
  type Point,
  uniqueIds,
} from "../geometry";

export type SliceDatum = { id: string; name?: string; value: number };
export type SliceShape = SliceDatum & {
  startAngle: number;
  endAngle: number;
  padAngle: number;
  // Unit vector from the center to the slice's middle (for the pull-out and labels).
  dir: Point;
  labelAt: Point;
};
export type PieLayout = { cx: number; cy: number; outer: number; inner: number; slices: SliceShape[] };

const CHART = "PieChart";

// Slices in data order, clockwise from 12 o'clock. A plain pie leaves room around the
// circle for outside labels on small slices.
export const pieLayout = (data: SliceDatum[], box: Box, donut = false): PieLayout => {
  checkCount(CHART, "slices", data.length, charts.limits.slices);
  checkValues(CHART, data.map((d) => d.value));
  uniqueIds(CHART, data.map((d) => d.id));
  if (sum(data, (d) => d.value) === 0) {
    throw new ChartLimitError(`${CHART}: values add up to 0`);
  }
  const cx = box.x + box.width / 2;
  const cy = box.y + box.height / 2;
  const outer = Math.min(box.width, box.height) / 2 - (donut ? charts.pull : charts.labelSize * 1.3);
  const inner = donut ? outer * 0.6 : 0;
  const labelR = donut ? (outer + inner) / 2 : outer * 0.62;
  const arcs = pie<SliceDatum>()
    .value((d) => d.value)
    .sort(null)
    .padAngle(0.015)(data);
  const slices = arcs.map((a) => {
    const mid = (a.startAngle + a.endAngle) / 2;
    const dir = { x: Math.sin(mid), y: -Math.cos(mid) };
    return {
      ...a.data,
      startAngle: a.startAngle,
      endAngle: a.endAngle,
      padAngle: a.padAngle,
      dir,
      labelAt: { x: cx + dir.x * labelR, y: cy + dir.y * labelR },
    };
  });
  return { cx, cy, outer, inner, slices };
};

// Path of one slice around (0, 0), cut off at `endAngle` for the sweep-in.
export const slicePath = (layout: PieLayout, slice: SliceShape, endAngle = slice.endAngle): string =>
  arc()({
    innerRadius: layout.inner,
    outerRadius: layout.outer,
    startAngle: slice.startAngle,
    endAngle: Math.max(slice.startAngle, Math.min(endAngle, slice.endAngle)),
    padAngle: slice.padAngle,
  }) ?? "";
```

```tsx
  chartProgress,

import { pieLayout, type SliceDatum, slicePath } from "./layout";

const SWEEP = durations.draw * 2;
// Slices narrower than this (radians) get no inside label.
const MIN_LABEL_ANGLE = 0.45;

export type PieChartProps = ChartBaseProps & { data: SliceDatum[]; donut?: boolean };

// Slices sweep in clockwise; the key slice pulls out and turns amber at keyAt.
// Pie: values inside the slices. Donut: the key value (big) and name in the hole.

  box,
  donut = false,
  keyId,
  values = "key",
  format,
  unit,
  ...timing

  const layout = pieLayout(data, box, donut);
  const at = timing.at ?? 0;
  const p = chartProgress(frame, timing);
  const labels = chartProgress(frame, { ...timing, at: at + SWEEP });
  const sweep = timing.still
    ? 1
    : interpolate(frame, [at, at + SWEEP], [0, 1], { ...clamp, easing: ease.smooth });

          return null;

              }}
            />
            {donut ? null : (
              <ChartText
                x={lx + s.dir.x * pull}
                y={ly + s.dir.y * pull}
                text={formatValue(s.value, format, unit)}
                color={big ? textOn(isKey, labels.key) : colors.text.secondary}
                opacity={big || isKey ? valueOpacity(values, isKey, labels) : 0}

            y={layout.cy - (key.name ? LINE * 0.4 : 0)}
            text={formatValue(key.value, format, unit)}

            size={typeScale.title}
            color={keyColor(true, labels.key, colors.text.primary)}

              x={layout.cx}
              y={layout.cy + LINE * 0.6}
              text={key.name}
              color={colors.text.muted}
              opacity={valueOpacity(values, true, labels)}
            />
          ) : null}
        </>
      ) : null}
```

```ts
import { hierarchy, treemap, treemapSquarify } from "d3-hierarchy";
import { charts } from "../../../brand/charts";
import { type Box, ChartLimitError, checkCount, checkValues, uniqueIds } from "../geometry";

// Shared by Treemap and TreeDiagram. Leaves carry `value` (treemap block size).
export type TreeDatum = { id: string; name?: string; value?: number; children?: TreeDatum[] };

export type BlockShape = {
  id: string;
  name?: string;
  value: number;
  depth: number;
  leaf: boolean;
  // Entrance order: parents first, then their children.
  order: number;
  x: number;
  y: number;
  width: number;
  height: number;
};

const CHART = "Treemap";

// Squarified treemap, biggest first. At most 2 levels below the root (groups + items).
export const treemapLayout = (root: TreeDatum, box: Box): BlockShape[] => {
  const h = hierarchy(root).sum((d) => (d.children?.length ? 0 : (d.value ?? 0)));
  const leaves = h.leaves();
  checkCount(CHART, "leaves", leaves.length, charts.limits.leaves);
  checkValues(CHART, leaves.map((l) => l.data.value ?? 0));
  uniqueIds(CHART, h.descendants().map((d) => d.data.id));
  if (h.height > 2) {
    throw new ChartLimitError(`${CHART}: ${h.height} levels below the root, max 2`);
  }
  if (!h.value) {
    throw new ChartLimitError(`${CHART}: values add up to 0`);
  }
  h.sort((a, b) => (b.value ?? 0) - (a.value ?? 0));
  const laid = treemap<TreeDatum>()
    .tile(treemapSquarify)
    .size([box.width, box.height])
    .paddingInner(10)
    .paddingOuter(6)
    .round(true)(h);
  return laid
    .descendants()
    .filter((d) => d.depth > 0)
    .map((d, order) => ({
      id: d.data.id,
      name: d.data.name,
      value: d.value ?? 0,
      depth: d.depth,
      leaf: !d.children,
      order,
      x: box.x + d.x0,
      y: box.y + d.y0,
      width: d.x1 - d.x0,
      height: d.y1 - d.y0,
    }));
};
```

```ts
import { hierarchy, tree } from "d3-hierarchy";
import { linkVertical } from "d3-shape";
import { charts } from "../../../brand/charts";
import { type Box, ChartLimitError, checkCount, checkFits, LINE, uniqueIds } from "../geometry";
import type { TreeDatum } from "../treemap/layout";

export const NODE_R = 22;

export type TreeNodeShape = { id: string; name?: string; depth: number; x: number; y: number; parentId?: string };
// One link per child node, keyed by the child's id.
export type TreeLinkShape = { id: string; depth: number; d: string };

const CHART = "TreeDiagram";

// Top-down tree; names under the nodes. Names must fit between their neighbours.
export const treeDiagramLayout = (
  root: TreeDatum,
  box: Box,
): { nodes: TreeNodeShape[]; links: TreeLinkShape[] } => {
  const h = hierarchy(root);
  uniqueIds(CHART, h.descendants().map((d) => d.data.id));
  if (h.height + 1 > charts.limits.treeDepth) {
    throw new ChartLimitError(`${CHART}: ${h.height + 1} levels, max ${charts.limits.treeDepth}`);
  }
  for (let depth = 0; depth <= h.height; depth++) {
    const count = h.descendants().filter((d) => d.depth === depth).length;
    checkCount(CHART, `nodes on level ${depth + 1}`, count, charts.limits.treeLevel);
  }

  const inset = charts.labelSize;
  const laid = tree<TreeDatum>().size([box.width - inset * 2, box.height - NODE_R * 2 - LINE])(h);
  const nodes = laid.descendants().map((n) => ({
    id: n.data.id,
    name: n.data.name,
    depth: n.depth,
    x: box.x + inset + n.x,
    y: box.y + NODE_R + n.y,
    parentId: n.parent?.data.id,
  }));

  for (const n of nodes) {
    if (!n.name) {
      continue;
    }
    const level = nodes.filter((o) => o.depth === n.depth && o.id !== n.id);
    const gap = Math.min(box.width, ...level.map((o) => Math.abs(o.x - n.x)));
    const edge = 2 * Math.min(n.x - box.x, box.x + box.width - n.x);
    checkFits(CHART, n.name, Math.min(gap - 16, edge));
  }

  const link = linkVertical();
  const pos = new Map(nodes.map((n) => [n.id, n]));
  const links = nodes
    .filter((n) => n.parentId !== undefined)
    .map((n) => {
      const parent = pos.get(n.parentId ?? "");
      return {
        id: n.id,
        depth: n.depth,
        d: link({ source: [parent?.x ?? n.x, parent?.y ?? n.y], target: [n.x, n.y] }) ?? "",
      };
    });
  return { nodes, links };
};

// Ids from the key node up to the root (empty when there's no key).
export const pathIds = (nodes: TreeNodeShape[], keyId: string | undefined): Set<string> => {
  const byId = new Map(nodes.map((n) => [n.id, n]));
  const ids = new Set<string>();
  let current = keyId === undefined ? undefined : byId.get(keyId);
  while (current) {
    ids.add(current.id);
    current = current.parentId === undefined ? undefined : byId.get(current.parentId);
  }
  return ids;
};
```

```ts
import { sankey, sankeyLinkHorizontal, type SankeyNode } from "d3-sankey";
import { charts } from "../../../brand/charts";
import {
  type Box,
  ChartLimitError,
  checkCount,
  checkFits,
  checkValues,
  type Point,
  uniqueIds,
} from "../geometry";

export type SankeyData = {
  nodes: { id: string; name?: string }[];
  links: { source: string; target: string; value: number }[];
};
export type SankeyNodeShape = {
  id: string;
  name?: string;
  value: number;
  column: number;
  x: number;
  y: number;
  width: number;
  height: number;
  labelAt: Point;
  anchor: "start" | "end";
};
export type SankeyLinkShape = {
  // `${source}->${target}`: use as keyId.
  id: string;
  value: number;
  column: number;
  d: string;
  width: number;
  mid: Point;
};

type N = { id: string; name?: string };
type L = Record<string, unknown>;

const CHART = "Sankey";
const NODE_W = 24;

// Flows left to right in 2-3 columns. Labels sit right of each node, except the last
// column (left of its nodes), and must fit in the gap between columns.
export const sankeyLayout = (
  data: SankeyData,
  box: Box,
): { nodes: SankeyNodeShape[]; links: SankeyLinkShape[] } => {
  uniqueIds(CHART, data.nodes.map((n) => n.id));
  checkCount(CHART, "links", data.links.length, Number.POSITIVE_INFINITY);
  checkValues(CHART, data.links.map((l) => l.value));
  const known = new Set(data.nodes.map((n) => n.id));
  for (const l of data.links) {
    for (const end of [l.source, l.target]) {
      if (!known.has(end)) {
        throw new ChartLimitError(`${CHART}: link uses unknown node "${end}"`);
      }
    }
  }

  const graph = sankey<N, L>()
    .nodeId((d) => d.id)
    .nodeWidth(NODE_W)
    .nodePadding(charts.labelSize * 1.6)
    .extent([
      [box.x, box.y],
      [box.x + box.width, box.y + box.height],
    ])({
    nodes: data.nodes.map((n) => ({ ...n })),
    links: data.links.map((l) => ({ ...l })),
  });

  const xs = [...new Set(graph.nodes.map((n) => n.x0 ?? 0))].sort((a, b) => a - b);
  if (xs.length < 2 || xs.length > charts.limits.sankeyColumns) {
    throw new ChartLimitError(
      `${CHART}: ${xs.length} columns, needs 2-${charts.limits.sankeyColumns}`,
    );
  }
  xs.forEach((x, column) => {
    const count = graph.nodes.filter((n) => n.x0 === x).length;
    checkCount(CHART, `nodes in column ${column + 1}`, count, charts.limits.sankeyColumn);
  });
  const gap = (box.width - NODE_W * xs.length) / (xs.length - 1);
  const last = xs.length - 1;

  const nodes = graph.nodes.map((n) => {
    const column = xs.indexOf(n.x0 ?? 0);
    const x0 = n.x0 ?? 0;
    const x1 = n.x1 ?? 0;
    const y0 = n.y0 ?? 0;
    const y1 = n.y1 ?? 0;
    if (n.name) {
      checkFits(CHART, n.name, gap - 24);
    }
    return {
      id: n.id,
      name: n.name,
      value: n.value ?? 0,
      column,
      x: x0,
      y: y0,
      width: x1 - x0,
      height: y1 - y0,
      anchor: column === last ? ("end" as const) : ("start" as const),
      labelAt: { x: column === last ? x0 - 12 : x1 + 12, y: (y0 + y1) / 2 },
    };
  });

  const path = sankeyLinkHorizontal<N, L>();
  const links = graph.links.map((l) => {
    const s = l.source as SankeyNode<N, L>;
    const t = l.target as SankeyNode<N, L>;
    return {
      id: `${s.id}->${t.id}`,
      value: l.value,
      column: xs.indexOf(s.x0 ?? 0),
      d: path(l) ?? "",
      width: l.width ?? 0,
      mid: { x: ((s.x1 ?? 0) + (t.x0 ?? 0)) / 2, y: ((l.y0 ?? 0) + (l.y1 ?? 0)) / 2 },
    };
  });
  return { nodes, links };
};
```

```tsx
            x={l.mid.x}
            y={l.mid.y}
```

```ts
import { max } from "d3-array";
import { scaleLinear } from "d3-scale";
import { charts } from "../../../brand/charts";
import {
  type Box,
  ChartLimitError,
  checkCount,
  checkFits,
  checkValues,
  LINE,
  type Point,
  uniqueIds,
} from "../geometry";

export type FunnelDatum = { id: string; name?: string; value: number };
export type StageShape = FunnelDatum & {
  // SVG polygon points.
  points: string;
  top: number;
  bottom: number;
  topWidth: number;
  bottomWidth: number;
  nameAt: Point;
  valueAt: Point;
};

const CHART = "Funnel";
const GAP = 8;

// Trapezoid stages, width by value, centered on the box. Name over value inside each stage.
export const funnelLayout = (data: FunnelDatum[], box: Box): StageShape[] => {
  checkCount(CHART, "stages", data.length, charts.limits.stages);
  checkValues(CHART, data.map((d) => d.value));
  uniqueIds(CHART, data.map((d) => d.id));
  const top = max(data, (d) => d.value) ?? 0;
  if (top === 0) {
    throw new ChartLimitError(`${CHART}: all values are 0`);
  }
  const n = data.length;
  const h = (box.height - GAP * (n - 1)) / n;
  const named = data.some((d) => d.name);
  if (h < (named ? LINE * 1.8 : LINE)) {
    throw new ChartLimitError(`${CHART}: stages are ${Math.round(h)} px tall; needs a taller box or fewer stages`);
  }
  const width = scaleLinear([0, top], [box.width * 0.18, box.width]);
  const cx = box.x + box.width / 2;

  return data.map((d, i) => {
    const y0 = box.y + i * (h + GAP);
    const y1 = y0 + h;
    const tw = width(d.value);
    const bw = Math.min(tw, i < n - 1 ? width(data[i + 1].value) : tw * 0.85);
    const mid = y0 + h / 2;
    if (d.name) {
      // Text sits at mid-height, where the stage is as wide as the average of its edges.
      checkFits(CHART, d.name, (tw + bw) / 2 - 24);
    }
    return {
      ...d,
      points: [
        [cx - tw / 2, y0],
```

```tsx
import { Interactive, useCurrentFrame } from "remotion";
import { charts } from "../../../brand/charts";
import { colors } from "../../../brand/tokens";
import {
  type ChartBaseProps,
  ChartText,
  chartProgress,
  keyColor,
  textOn,
  valueOpacity,
} from "../common";
import { formatValue } from "../geometry";
import { type FunnelDatum, funnelLayout } from "./layout";

export type FunnelProps = ChartBaseProps & { data: FunnelDatum[] };

// Stages drop in top to bottom; the key stage turns amber at keyAt.
// SVG, for SceneCanvas / SlideCanvas.
export const Funnel: React.FC<FunnelProps> = ({
  data,
  box,
  keyId,
  values = "key",
  format,
  unit,
  ...timing
}) => {
  const frame = useCurrentFrame();
  const stages = funnelLayout(data, box);
  const base = chartProgress(frame, timing);

  return (
    <Interactive.G name="Funnel" style={{ opacity: base.exit }}>
      {stages.map((s, i) => {
        const p = chartProgress(frame, timing, i);
        const isKey = s.id === keyId;
        return (
          <g key={s.id} style={{ opacity: p.enter, translate: `0px ${(1 - p.enter) * -30}px` }}>
            <polygon
              points={s.points}
              fill={keyColor(isKey, p.key, colors.diagram.fill)}
              stroke={keyColor(isKey, p.key, colors.diagram.soft)}
              strokeWidth={charts.stroke}
              strokeLinejoin="round"
            />
            {s.name ? (
              <ChartText
                x={s.nameAt.x}
                y={s.nameAt.y}
                text={s.name}
                color={isKey && p.key > 0.5 ? colors.bg.base : colors.text.muted}
              />
            ) : null}
            <ChartText
              x={s.valueAt.x}
              y={s.valueAt.y}
              text={formatValue(s.value, format, unit)}
              color={textOn(isKey, p.key)}
              opacity={valueOpacity(values, isKey, p)}
            />
          </g>
        );
      })}
    </Interactive.G>
  );
};
```
